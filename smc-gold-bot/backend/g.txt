function stripDataUrlPrefix(dataUrl) {
  const commaIdx = dataUrl.indexOf(',');
  return commaIdx >= 0 ? dataUrl.slice(commaIdx + 1) : dataUrl;
}

function safeParseJson(text) {
  const cleaned = text.replace(/```json|```/g, '').trim();
  return JSON.parse(cleaned);
}

function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

async function fetchWithRetry(url, options, retries = 2) {
  for (let attempt = 0; attempt <= retries; attempt++) {
    const res = await fetch(url, options);

    // فقط برای خطاهای موقت سمت گوگل (503 شلوغی، 429 محدودیت نرخ) دوباره امتحان کن
    if ((res.status === 503 || res.status === 429) && attempt < retries) {
      const delay = 1500 * (attempt + 1); // 1.5s, 3s
      await sleep(delay);
      continue;
    }

    return res;
  }
}

async function analyzeChart(chartFrameDataUrl, currentPrice, m5Candles = []) {
  if (!GEMINI_API_KEY) {
    throw new Error('GEMINI_API_KEY تنظیم نشده است');
  }

  const base64Image = stripDataUrlPrefix(chartFrameDataUrl);

  const priceText = currentPrice
    ? `قیمت لحظه‌ای فعلی XAUUSD: ${currentPrice}`
    : 'قیمت لحظه‌ای در دسترس نیست.';

  const m5Sample = m5Candles.slice(-30).map((c) => ({
    t: c.time, o: c.open, h: c.high, l: c.low, c: c.close,
  }));

  const userText = `${priceText}

کندل‌های M5 اخیر (برای تأیید نقطه ورود، جدیدترین در انتها):
${JSON.stringify(m5Sample)}`;

  const url = `https://generativelanguage.googleapis.com/v1beta/models/${GEMINI_MODEL}:generateContent?key=${GEMINI_API_KEY}`;

  const res = await fetchWithRetry(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      contents: [
        {
          role: 'user',
          parts: [
            { text: `${SYSTEM_PROMPT}\n\n${userText}` },
            { inline_data: { mime_type: 'image/jpeg', data: base64Image } },
          ],
        },
      ],
      generationConfig: {
        temperature: 0.2,
        response_mime_type: 'application/json',
      },
    }),
  });

  if (!res.ok) {
    const errBody = await res.text().catch(() => '');
    throw new Error(`خطای Gemini API (${res.status}): ${errBody.slice(0, 200)}`);
  }

  const data = await res.json();
  const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;

  if (!text) {
    throw new Error('پاسخ Gemini خالی یا نامعتبر بود');
  }

  const parsed = safeParseJson(text);

  return {
    direction: (parsed.direction || 'wait').toLowerCase(),
    entry: Number(parsed.entry) || 0,
    stop_loss: Number(parsed.stop_loss) || 0,
    take_profit: Number(parsed.take_profit) || 0,
    confidence: Math.max(0, Math.min(1, Number(parsed.confidence) || 0)),
    reason: parsed.reason || '',
  };
}

module.exports = { analyzeChart };
